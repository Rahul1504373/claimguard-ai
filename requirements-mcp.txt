# Dependencies for mcp_server.py ONLY — install into a SEPARATE venv from
# requirements.txt (the FastAPI/Streamlit app's dependencies).
#
# WHY SEPARATE: the `mcp` SDK pulls in a newer `starlette` (and `pydantic`
# via mcp-types) than requirements.txt pins for FastAPI compatibility.
# Installing both in one environment breaks the FastAPI app — confirmed
# directly during development (see mcp_server.py's module docstring and
# docs/tracker.md for the exact error). This is a real, stated dependency
# conflict, not a hypothetical one.
#
#   python3 -m venv venv-mcp
#   source venv-mcp/bin/activate
#   pip install -r requirements-mcp.txt
#   python3 mcp_server.py
#
# mcp_server.py imports app/storage.py, app/config.py, and
# app/agents/retrieval_agent.py directly — none of which depend on FastAPI
# or Pydantic, so running them under this separate venv's newer dependency
# versions is safe.
mcp==2.3.0
python-dotenv==1.0.1
scikit-learn==1.5.2
numpy==2.1.2
tenacity==8.5.0

# Only needed to run tests/test_mcp_server.py (exercises the pipeline to
# seed real claim data) — not needed to just run mcp_server.py itself.
